# Define the Context class
class Context:
    def __init__(self, year=None, organization=None, project=None):
        self.year = year
        self.organization = organization
        self.project = project

    def __repr__(self):
        return f"Context(year='{self.year}', organization='{self.organization}', project='{self.project}')"

    def __eq__(self, other):
        if not isinstance(other, Context):
            return False
        return (self.year == other.year and 
                self.organization == other.organization and 
                self.project == other.project)

def main():
    """Main function to parse arguments and generate license"""
    parser = ArgumentParser(description="Generate license files")
    parser.add_argument("license", nargs='?', default="mit", help="License type")
    parser.add_argument("-o", "--org", dest="organization", help="Organization name")

    parser.add_argument("-t", "--template", dest="template_path", help="Custom template path")
    parser.add_argument("-y", "--year", dest="year", help="Copyright year")
    parser.add_argument("-l", "--language", dest="language", help="Language for header formatting")

    context = get_context(args)
    
    # If --vars flag is used, list variables and exit

    # Load template
    template = load_template(args.license, bool(args.language))
    
    # Generate license
    license_text = generate_license(template, context)

    if args.output_file:

        print(license_text.getvalue())

def clean_path(p):
    """Clean up a path string"""
    return os.path.normpath(p)

def get_context(args):
    """Create context object from parsed arguments"""
    # Determine year
    year = args.year or str(os.environ.get('YEAR', '2023'))
    
    # Determine organization
    org = args.organization or guess_organization()
    
    # Determine project
    project = args.project or os.environ.get('PROJECT', '')
    
    return Context(year=year, organization=org, project=project)

def guess_organization():
    """Guess organization from git config or USER env var"""
    try:
        import subprocess
        result = subprocess.run(['git', 'config', '--global', 'user.name'], 
                              capture_output=True, text=True, check=True)
        return result.stdout.strip()
    except (subprocess.CalledProcessError, FileNotFoundError):
        return os.environ.get('USER', 'Unknown')

def load_file_template(path):
    """Load a template from a file"""
    with open(path, 'r') as f:
        content = f.read()
    return StringIO(content)

def load_template(license_type, header=False):
    """Load a template based on license type and header flag"""
    # Determine template name
    template_name = f"template-{license_type}{'-header' if header else ''}.txt"
    
    # Try to load from current directory first, then fall back to installed templates
    template_path = clean_path(f"./templates/{template_name}")
    
    if os.path.exists(template_path):
        return load_file_template(template_path)
    
    # Fall back to installed templates (this would require proper packaging)
    # For now, we'll just raise an error
    raise FileNotFoundError(f"Template not found: {template_name}")

def extract_vars(template):
    """Extract variables from template"""
    template.seek(0)
    lines = template.readlines()
    vars_list = []
    for line in lines:
        if '{' in line and '}' in line:
            # Simple extraction of variables between braces
            start = 0
            while True:
                start = line.find('{', start)
                if start == -1:
                    break
                end = line.find('}', start)
                if end != -1:
                    var = line[start+1:end]
                    if var not in vars_list:
                        vars_list.append(var)
                    start = end + 1
    return vars_list

def generate_license(template, context):
    """Generate license text with context"""
    template.seek(0)
    content = template.read()
    
    # Replace variables with context values
    result = content.format(
        year=context.year,
        organization=context.organization,
        project=context.project
    )
    
    return StringIO(result)
